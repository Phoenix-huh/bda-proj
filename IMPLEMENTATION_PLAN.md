## 3. Phase 1: Data Cache (about 2 hours)

Problem: `app.py` loads the 222 MB CSV (2.59M rows) at startup, which takes 35 s and over 1.5 GB of RAM.
Fix: pre-aggregate once into small files and an indexed SQLite table.

### 3.1 Create `build_cache.py`

```python
"""
One-off script: reads the big hourly CSV in chunks and writes small cache files.

Usage:  python build_cache.py
Output: cache/global_means.json
        cache/station_means.csv
        cache/station_hour_means.csv
        cache/station_daily.csv
        cache/hourly.db  (SQLite, indexed on StationId + Datetime)
"""
import json
import os
import sqlite3
import time

import numpy as np
import pandas as pd

SRC = os.path.join('data', 'station_hour_cleaned.csv')
OUT = 'cache'
TARGETS = ['PM2.5', 'PM10', 'NO', 'NO2', 'NOx', 'NH3', 'CO', 'SO2',
           'O3', 'Benzene', 'Toluene', 'Xylene', 'AQI']
CHUNK = 250_000


def main():
    if not os.path.exists(SRC):
        raise FileNotFoundError(f"{SRC} not found. Run from the project root.")
    os.makedirs(OUT, exist_ok=True)

    db_path = os.path.join(OUT, 'hourly.db')
    if os.path.exists(db_path):
        os.remove(db_path)
    con = sqlite3.connect(db_path)

    daily_sum, daily_cnt = [], []   # keyed by (StationId, Date)
    hour_sum, hour_cnt = [], []     # keyed by (StationId, Hour)

    start = time.time()
    total = 0
    for chunk in pd.read_csv(SRC, usecols=['StationId', 'Datetime'] + TARGETS,
                             chunksize=CHUNK, low_memory=False):
        dt = pd.to_datetime(chunk['Datetime'], utc=True, errors='coerce')
        chunk = chunk[dt.notna()].copy()
        dt = dt[dt.notna()]
        chunk['Date'] = dt.dt.strftime('%Y-%m-%d')
        chunk['Hour'] = dt.dt.hour
        chunk['Datetime'] = dt.dt.strftime('%Y-%m-%d %H:00:00')

        g = chunk.groupby(['StationId', 'Date'])[TARGETS]
        daily_sum.append(g.sum())
        daily_cnt.append(g.count())

        g = chunk.groupby(['StationId', 'Hour'])[TARGETS]
        hour_sum.append(g.sum())
        hour_cnt.append(g.count())

        chunk[['StationId', 'Datetime'] + TARGETS].to_sql(
            'hourly', con, if_exists='append', index=False)

        total += len(chunk)
        print(f"  processed {total:,} rows ({time.time() - start:.0f}s)")

    print("Creating SQLite index...")
    con.execute('CREATE INDEX IF NOT EXISTS idx_station_dt ON hourly (StationId, Datetime)')
    con.commit()
    con.close()

    def combine(sums, cnts):
        s = pd.concat(sums).groupby(level=list(range(pd.concat(sums).index.nlevels))).sum()
        c = pd.concat(cnts).groupby(level=list(range(pd.concat(cnts).index.nlevels))).sum()
        return s, c

    d_s, d_c = combine(daily_sum, daily_cnt)
    h_s, h_c = combine(hour_sum, hour_cnt)

    # Daily means per station (for line charts)
    daily = (d_s / d_c.replace(0, np.nan)).reset_index()
    daily.to_csv(os.path.join(OUT, 'station_daily.csv'), index=False)

    # Per-station means (for bar/pie charts and lag fallback)
    st_s = d_s.groupby(level=0).sum()
    st_c = d_c.groupby(level=0).sum()
    station_means = (st_s / st_c.replace(0, np.nan)).reset_index()
    station_means.to_csv(os.path.join(OUT, 'station_means.csv'), index=False)

    # Per-station-per-hour-of-day means (better lag fallback)
    station_hour_means = (h_s / h_c.replace(0, np.nan)).reset_index()
    station_hour_means.to_csv(os.path.join(OUT, 'station_hour_means.csv'), index=False)

    # Global means (last-resort fallback)
    global_means = (d_s.sum() / d_c.sum().replace(0, np.nan)).to_dict()
    with open(os.path.join(OUT, 'global_means.json'), 'w') as f:
        json.dump({k: float(v) for k, v in global_means.items()}, f, indent=2)

    print(f"Done in {time.time() - start:.0f}s. Rows: {total:,}")


if __name__ == '__main__':
    main()
```

## 4. Phase 2: Fix Training and Rewrite the App (about 4 hours)

### 4.1 Problems being fixed

| Bug | Fix |
|-----|-----|
| BUG-05 static national means used as lag features | Lag lookup: observed reading → station-hour mean → station mean → global mean |
| Lag features computed AFTER sampling (an extra bug not in the original report) | Compute lags on the full data, THEN sample rows |
| Inference doesn't apply training's clipping | Save `models/clip_bounds.json` and apply at inference |
| BUG-08 deprecated `fillna(method=...)` | Removed (rows with bad datetimes are dropped) |
| BUG-11 unused `stations` load | Removed |
| Hard-coded `sample_frac=0.0386` | `--sample-frac` CLI argument |
| BUG-06 intractable SVR | `--model linearsvr` uses scaled `LinearSVR` |
| Unlabelled/falsified metrics | Metrics JSON now records the real model name and row count |

**Note on the lag bug:** the original script sampled 3.86% of rows and then computed `shift(1)`. After sampling, "the previous row" is a random earlier hour, not the previous hour. Lags are meaningless that way. The new script computes lags on the complete sorted data first.

### 4.2 Rewrite `train_model.py`

```python
"""
Train multi-output regression for air-quality pollutants.

Usage:
    python train_model.py                          # Ridge, ~4% sample
    python train_model.py --sample-frac 0.2        # Ridge, 20% sample
    python train_model.py --model linearsvr        # scaled LinearSVR
"""
import argparse
import json
import logging
import os
import time

import joblib
import numpy as np
import pandas as pd
from sklearn.compose import ColumnTransformer, TransformedTargetRegressor
from sklearn.linear_model import Ridge
from sklearn.metrics import r2_score
from sklearn.model_selection import KFold
from sklearn.multioutput import MultiOutputRegressor
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import OneHotEncoder, StandardScaler
from sklearn.svm import LinearSVR

logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s',
    handlers=[logging.FileHandler('train_model.log'), logging.StreamHandler()],
)
logger = logging.getLogger(__name__)

DATA_PATH = os.path.join('data', 'station_hour_cleaned.csv')
TARGETS = ['PM2.5', 'PM10', 'NO', 'NO2', 'NOx', 'NH3', 'CO', 'SO2',
           'O3', 'Benzene', 'Toluene', 'Xylene', 'AQI']
LAGS = [f'{t}_lag1' for t in TARGETS]
TIME_FEATURES = ['Year', 'Month', 'Day', 'Hour']
FEATURES = ['StationId'] + TIME_FEATURES + LAGS

MODEL_FILES = {
    'ridge':     ('Ridge',     'models/air_quality_model.pkl',     'models/accuracy_metrics.json'),
    'linearsvr': ('LinearSVR', 'models/air_quality_model_svm.pkl', 'models/accuracy_metrics_svm.json'),
}


def load_and_build_features(path):
    """Load full data, build lag features on the COMPLETE sorted series."""
    if not os.path.exists(path):
        raise FileNotFoundError(f"File not found: {path}")
    df = pd.read_csv(path, usecols=['StationId', 'Datetime'] + TARGETS, low_memory=False)
    logger.info(f"Loaded {len(df):,} rows")

    df['Datetime'] = pd.to_datetime(df['Datetime'], errors='coerce')
    bad = df['Datetime'].isna().sum()
    if bad:
        logger.warning(f"Dropping {bad} rows with unparseable Datetime")
        df = df.dropna(subset=['Datetime'])

    df = df.sort_values(['StationId', 'Datetime']).reset_index(drop=True)

    means = df[TARGETS].mean()
    grouped = df.groupby('StationId')
    for t in TARGETS:
        df[f'{t}_lag1'] = grouped[t].shift(1).fillna(means[t])

    df['Year'] = df['Datetime'].dt.year
    df['Month'] = df['Datetime'].dt.month
    df['Day'] = df['Datetime'].dt.day
    df['Hour'] = df['Datetime'].dt.hour
    return df.drop(columns=['Datetime'])


def clean_numeric(data):
    """Median-fill NaNs and clip targets/lags to the 1st-99th percentile. Returns bounds for lags."""
    bounds = {}
    for col in TARGETS + LAGS:
        data[col] = data[col].fillna(data[col].median())
        low, high = data[col].quantile([0.01, 0.99])
        data[col] = data[col].clip(low, high)
        if col in LAGS:
            bounds[col] = [float(low), float(high)]
    return data, bounds


def build_model(kind):
    cat = ('cat', OneHotEncoder(handle_unknown='ignore'), ['StationId'])
    if kind == 'ridge':
        pre = ColumnTransformer([cat], remainder='passthrough')
        reg = Ridge(alpha=1.0)
    else:
        num = ('num', StandardScaler(), TIME_FEATURES + LAGS)
        pre = ColumnTransformer([cat, num])
        reg = MultiOutputRegressor(
            TransformedTargetRegressor(
                regressor=LinearSVR(epsilon=0.1, C=1.0, dual='auto', max_iter=5000, random_state=42),
                transformer=StandardScaler(),
            )
        )
    return Pipeline([('preprocessor', pre), ('regressor', reg)])


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--model', choices=list(MODEL_FILES), default='ridge')
    parser.add_argument('--sample-frac', type=float, default=0.0386,
                        help='fraction of rows used for training (1.0 = all)')
    parser.add_argument('--folds', type=int, default=5)
    args = parser.parse_args()

    name, model_path, metrics_path = MODEL_FILES[args.model]
    os.makedirs('models', exist_ok=True)

    data = load_and_build_features(DATA_PATH)
    if args.sample_frac < 1.0:
        data = data.sample(frac=args.sample_frac, random_state=42)
        logger.info(f"Sampled {len(data):,} rows ({args.sample_frac * 100:.2f}%)")

    data, bounds = clean_numeric(data)
    X, y = data[FEATURES], data[TARGETS]

    model = build_model(args.model)
    kf = KFold(n_splits=args.folds, shuffle=True, random_state=42)
    scores = {t: [] for t in TARGETS}

    start = time.time()
    for fold, (tr, te) in enumerate(kf.split(X), 1):
        model.fit(X.iloc[tr], y.iloc[tr])
        pred = model.predict(X.iloc[te])
        for i, t in enumerate(TARGETS):
            scores[t].append(r2_score(y.iloc[te][t], pred[:, i]))
        logger.info(f"Fold {fold}/{args.folds} done ({time.time() - start:.0f}s)")

    mean_r2 = {t: float(np.mean(v)) for t, v in scores.items()}
    logger.info(f"Mean R2: {mean_r2}")

    logger.info("Fitting final model on all sampled rows...")
    model.fit(X, y)
    joblib.dump(model, model_path)
    logger.info(f"Saved model -> {model_path}")

    with open(metrics_path, 'w') as f:
        json.dump({'model': name, 'n_rows': int(len(data)), 'cv_folds': args.folds,
                   'sample_frac': args.sample_frac, 'r2': mean_r2}, f, indent=4)
    logger.info(f"Saved metrics -> {metrics_path}")

    if args.model == 'ridge':   # the app uses the Ridge model's bounds
        with open('models/clip_bounds.json', 'w') as f:
            json.dump(bounds, f, indent=2)
        logger.info("Saved clip bounds -> models/clip_bounds.json")


if __name__ == '__main__':
    main()
```

### 4.3 Rewrite `app.py`

Key changes:
- Loads only small cache files, so startup is under a second.
- Lag features come from a fallback chain.
- Charts use matplotlib's `Figure` object directly (no `pyplot` global state), so there are no leaks and it is thread-safe.
- Proper HTTP status codes (400/404/500) instead of 200 with an error body.
- `debug` is controlled by an environment variable.
- Dead commented-out code is gone.

```python
import base64
import datetime
import json
import logging
import os
import sqlite3
from io import BytesIO

import joblib
import matplotlib
matplotlib.use('Agg')
from matplotlib.figure import Figure
import pandas as pd
from flask import Flask, jsonify, render_template, request

BASE = os.path.dirname(os.path.abspath(__file__))


def path(*parts):
    return os.path.join(BASE, *parts)


MODEL_PATH = os.environ.get('MODEL_PATH', path('models', 'air_quality_model.pkl'))
DB_PATH = path('cache', 'hourly.db')

TARGETS = ['PM2.5', 'PM10', 'NO', 'NO2', 'NOx', 'NH3', 'CO', 'SO2',
           'O3', 'Benzene', 'Toluene', 'Xylene', 'AQI']

logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s',
    handlers=[logging.FileHandler(path('app.log')), logging.StreamHandler()],
)
logger = logging.getLogger(__name__)

app = Flask(__name__)

# ----------------------------------------------------------------------
# Startup: load model + small cache files only (no 222 MB CSV)
# ----------------------------------------------------------------------
model = joblib.load(MODEL_PATH)
logger.info(f"Model loaded from {MODEL_PATH}")

stations = pd.read_csv(path('data', 'stations.csv'))
station_list = [
    {'StationId': r.StationId, 'display': f"{r.City}, {r.State} ({r.StationId})"}
    for r in stations[['StationId', 'City', 'State']].dropna().itertuples()
]
valid_stations = {s['StationId'] for s in station_list}

with open(path('cache', 'global_means.json')) as f:
    global_means = json.load(f)

station_means = pd.read_csv(path('cache', 'station_means.csv')).set_index('StationId').to_dict('index')
station_hour_means = (pd.read_csv(path('cache', 'station_hour_means.csv'))
                      .set_index(['StationId', 'Hour']).to_dict('index'))
_daily = pd.read_csv(path('cache', 'station_daily.csv'), parse_dates=['Date'])
daily_by_station = {sid: g.set_index('Date').sort_index() for sid, g in _daily.groupby('StationId')}

clip_bounds = {}
if os.path.exists(path('models', 'clip_bounds.json')):
    with open(path('models', 'clip_bounds.json')) as f:
        clip_bounds = json.load(f)
else:
    logger.warning("models/clip_bounds.json missing; retrain with train_model.py")

logger.info(f"Ready: {len(valid_stations)} stations")


# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------
def error(msg, code=400):
    return jsonify({'error': msg}), code


def aqi_bucket(aqi):
    if aqi <= 50:
        return 'Good'
    if aqi <= 100:
        return 'Satisfactory'
    if aqi <= 200:
        return 'Moderate'
    if aqi <= 300:
        return 'Poor'
    if aqi <= 400:
        return 'Very Poor'
    return 'Severe'


def observed_previous_hour(station, when):
    """Actual reading at (station, when - 1h) from SQLite, or {}."""
    if not os.path.exists(DB_PATH):
        return {}
    prev = (when - datetime.timedelta(hours=1)).strftime('%Y-%m-%d %H:00:00')
    cols = ','.join(f'"{t}"' for t in TARGETS)
    con = sqlite3.connect(DB_PATH)
    try:
        row = con.execute(
            f'SELECT {cols} FROM hourly WHERE StationId=? AND Datetime=? LIMIT 1',
            (station, prev)).fetchone()
    finally:
        con.close()
    return dict(zip(TARGETS, row)) if row else {}


def build_lag_features(station, when):
    """Best available lag-1 values: observed > station-hour mean > station mean > global mean."""
    prev_hour = (when.hour - 1) % 24
    layers = [
        ('observed', observed_previous_hour(station, when)),
        ('station-hour-average', station_hour_means.get((station, prev_hour), {})),
        ('station-average', station_means.get(station, {})),
        ('global-average', global_means),
    ]
    values, used = {}, []
    for t in TARGETS:
        for name, source in layers:
            v = source.get(t)
            if v is not None and not pd.isna(v):
                col = f'{t}_lag1'
                v = float(v)
                if col in clip_bounds:
                    lo, hi = clip_bounds[col]
                    v = min(max(v, lo), hi)
                values[col] = v
                if name not in used:
                    used.append(name)
                break
    return values, used


def fig_to_b64(fig):
    buf = BytesIO()
    fig.savefig(buf, format='png', bbox_inches='tight')
    return base64.b64encode(buf.getvalue()).decode('ascii')


# ----------------------------------------------------------------------
# Routes
# ----------------------------------------------------------------------
@app.route('/')
def home():
    return render_template('index.html', stations=station_list, targets=TARGETS)


@app.route('/predict', methods=['POST'])
def predict():
    station = request.form.get('station', '').strip()
    date_str = request.form.get('date', '').strip()
    hour_str = request.form.get('hour', '').strip()

    if station not in valid_stations:
        return error(f"Unknown station: {station!r}")
    try:
        hour = int(hour_str)
        if not 0 <= hour <= 23:
            raise ValueError
    except ValueError:
        return error('Hour must be an integer between 0 and 23')
    try:
        when = datetime.datetime.strptime(date_str, '%Y-%m-%d').replace(hour=hour)
    except ValueError:
        return error('Date must be in YYYY-MM-DD format (e.g., 2019-01-15)')

    try:
        lags, used = build_lag_features(station, when)
        row = {'StationId': [station], 'Year': [when.year], 'Month': [when.month],
               'Day': [when.day], 'Hour': [hour]}
        row.update({k: [v] for k, v in lags.items()})
        pred = model.predict(pd.DataFrame(row))[0]
        predictions = {t: float(v) for t, v in zip(TARGETS, pred)}
        return jsonify({
            'predictions': predictions,
            'AQI_Bucket': aqi_bucket(predictions['AQI']),
            'lag_source': used,
        })
    except Exception:
        logger.exception("Prediction error")
        return error('Prediction failed. See server log.', 500)


@app.route('/visualize', methods=['POST'])
def visualize():
    station = request.form.get('station', '').strip()
    params = request.form.getlist('parameters') or list(TARGETS)

    if station not in valid_stations:
        return error(f"Unknown station: {station!r}")
    bad = [p for p in params if p not in TARGETS]
    if bad:
        return error(f"Unknown parameters: {bad}")
    if station not in station_means or station not in daily_by_station:
        return error(f"No data available for station {station}", 404)

    try:
        averages = pd.Series(station_means[station])[params].dropna()
        positive = averages[averages > 0]
        daily = daily_by_station[station]

        # Pie
        fig = Figure(figsize=(7, 6))
        ax = fig.subplots()
        ax.pie(positive.values, labels=positive.index, autopct='%1.1f%%')
        ax.set_title('Distribution of Selected Gases')
        pie_url = fig_to_b64(fig)

        # Bar
        fig = Figure(figsize=(8, 5))
        ax = fig.subplots()
        ax.bar(averages.index, averages.values, color='#66b3ff')
        ax.set_title('Average Values of Selected Gases')
        ax.set_ylabel('Average value')
        ax.set_xlabel('Parameter')
        fig.autofmt_xdate()
        bar_url = fig_to_b64(fig)

        # Line
        fig = Figure(figsize=(10, 5))
        ax = fig.subplots()
        for p in params:
            ax.plot(daily.index, daily[p], label=p, linewidth=1)
        ax.set_title('Daily Trend of Selected Gases')
        ax.set_ylabel('Value')
        ax.set_xlabel('Date')
        ax.legend(loc='upper right', ncol=2, fontsize=8)
        fig.autofmt_xdate()
        line_url = fig_to_b64(fig)

        return jsonify({'pie_url': pie_url, 'bar_url': bar_url, 'line_url': line_url})
    except Exception:
        logger.exception("Visualization error")
        return error('Visualization failed. See server log.', 500)


if __name__ == '__main__':
    app.run(debug=os.environ.get('FLASK_DEBUG') == '1')
```

## 8. Known Limitations (Document These)

1. **Future dates are not truly forecast.** The data covers 2015 to 2020. For dates outside it, the observed lag is unavailable, so the model falls back to station-hour averages. Predictions for 2026 are therefore essentially "typical level for this station, month and hour". For real forecasting, retrain without lag features or use recursive multi-step prediction.
2. **Lag-1 assumes consecutive hourly rows.** Gaps in the data can make `shift(1)` refer to an earlier time. A stricter version would reindex each station to a full hourly grid before shifting.
3. **Linear models.** Ridge and LinearSVR can't capture strong non-linear or seasonal effects. Gradient boosting (LightGBM or HistGradientBoostingRegressor) would likely score higher.
4. **AQI is predicted directly** as another target rather than computed from sub-index formulas, so its pollutant predictions and AQI may not be mutually consistent.
5. **Sampling.** Training uses about 4% of rows by default. Raise `--sample-frac` for better accuracy (Ridge handles 1.0 easily; LinearSVR will be slow).
6. **The Flask development server** is not for production. Use gunicorn or waitress behind a reverse proxy if deploying.
